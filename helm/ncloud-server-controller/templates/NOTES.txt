1. Get the application URL by running these commands:
{{- if contains "NodePort" .Values.metrics.service.type }}
  export NODE_PORT=$(kubectl get --namespace {{ .Values.namespace.name }} -o jsonpath="{.spec.ports[0].nodePort}" services {{ include "ncloud-server-controller.fullname" . }}-metrics-service)
  export NODE_IP=$(kubectl get nodes --namespace {{ .Values.namespace.name }} -o jsonpath="{.items[0].status.addresses[0].address}")
  echo http://$NODE_IP:$NODE_PORT
{{- else if contains "LoadBalancer" .Values.metrics.service.type }}
     NOTE: It may take a few minutes for the LoadBalancer IP to be available.
           You can watch the status of by running 'kubectl get --namespace {{ .Values.namespace.name }} svc -w {{ include "ncloud-server-controller.fullname" . }}-metrics-service'
  export SERVICE_IP=$(kubectl get svc --namespace {{ .Values.namespace.name }} {{ include "ncloud-server-controller.fullname" . }}-metrics-service --template "{{"{{ range (index .status.loadBalancer.ingress 0) }}{{.}}{{ end }}"}}")
  echo http://$SERVICE_IP:{{ .Values.metrics.service.port }}
{{- else if contains "ClusterIP" .Values.metrics.service.type }}
  export POD_NAME=$(kubectl get pods --namespace {{ .Values.namespace.name }} -l "app.kubernetes.io/name={{ include "ncloud-server-controller.name" . }},app.kubernetes.io/instance={{ .Release.Name }}" -o jsonpath="{.items[0].metadata.name}")
  export CONTAINER_PORT=$(kubectl get pod --namespace {{ .Values.namespace.name }} $POD_NAME -o jsonpath="{.spec.containers[0].ports[0].containerPort}")
  echo "Visit http://127.0.0.1:8080 to use your application"
  kubectl --namespace {{ .Values.namespace.name }} port-forward $POD_NAME 8080:$CONTAINER_PORT
{{- end }}

2. Check the operator status:
  kubectl get pods -n {{ .Values.namespace.name }} -l "app.kubernetes.io/name={{ include "ncloud-server-controller.name" . }}"

3. View the operator logs:
  kubectl logs -n {{ .Values.namespace.name }} -l "app.kubernetes.io/name={{ include "ncloud-server-controller.name" . }}" -f

4. Create a sample NCloudServer resource:
  kubectl apply -f - <<EOF
  apiVersion: server.ncloud.devops.ai.kr/v1
  kind: NCloudServer
  metadata:
    name: example-server
    namespace: {{ .Values.namespace.name }}
  spec:
    vpcNo: "vpc-12345"
    subnetNo: "subnet-12345"
    serverImageProductCode: "SPSW0LINUX000046"
    serverProductCode: "SPSVRSTAND000004"
    loginKeyName: "my-key"
    credentials:
      secretRef:
        name: ncloud-credentials
  EOF

5. Check NCloudServer resources:
  kubectl get ncloudservers -n {{ .Values.namespace.name }}

6. For more information about this operator, visit:
   https://github.com/Seo-yul/ncloud-server-controller
